1use thiserror::Error;

3/// Represents an error returned by TrajoptLib. Primarily exposes the Sleipnir
4/// [solver exit status](https://github.com/SleipnirGroup/Sleipnir/blob/v0.5.0/include/sleipnir/optimization/solver/exit_status.hpp).
5#[derive(Debug, Error)]
6pub enum TrajoptError {
7    #[error("Too few degrees of freedom")]
8    /// The solver determined the problem to be overconstrained and gave up
9    TooFewDOFs,
10    #[error("Locally infeasible")]
11    /// The solver determined the problem to be locally infeasible and gave up
12    LocallyInfeasible,
13    #[error("Globally infeasible")]
14    /// The solver determined the problem to be globally infeasible and gave up
15    GloballyInfeasible,
16    #[error("Factorization failed")]
17    /// The solver failed to factorize a matrix and gave up
18    FactorizationFailed,
19    #[error("Line search failed")]
20    /// The solver failed to converge with its line search and gave up
21    LineSearchFailed,
22    #[error("Feasibility restoration failed")]
23    /// The solver failed to reach the desired tolerance, and feasibility
24    /// restoration failed to converge
25    FeasibilityRestorationFailed,
26    #[error("Nonfinite initial guess")]
27    /// The solver encountered a nonfinite initial guess and gave up
28    NonfiniteInitialGuess,
29    #[error("Diverging iterates")]
30    /// The solver encountered diverging primal iterates xₖ and/or sₖ and gave
31    /// up
32    DivergingIterates,
33    #[error("Max iterations exceeded")]
34    /// The solver returned its solution so far after exceeding the maximum
35    /// number of iterations
36    MaxIterationsExceeded,
37    #[error("Timeout")]
38    /// The solver returned its solution so far after exceeding the maximum
39    /// elapsed wall clock time
40    Timeout,
41    #[error("Unparsable error code: {0}")]
42    /// The solver returned an unparsable error code
43    Unparsable(Box<str>),
44    #[error("Unknown error: {0:?}")]
45    /// Unknown error type
46    Unknown(i8),
47}
48
49impl From<i8> for TrajoptError {
50    fn from(value: i8) -> Self {
51        match value {
52            -1 => Self::TooFewDOFs,
53            -2 => Self::LocallyInfeasible,
54            -3 => Self::GloballyInfeasible,
55            -4 => Self::FactorizationFailed,
56            -5 => Self::LineSearchFailed,
57            -6 => Self::FeasibilityRestorationFailed,
58            -7 => Self::NonfiniteInitialGuess,
59            -8 => Self::DivergingIterates,
60            -9 => Self::MaxIterationsExceeded,
61            -10 => Self::Timeout,
62            _ => Self::Unknown(value),
63        }
64    }
65}

72        format!("{self}").serialize(serializer)
