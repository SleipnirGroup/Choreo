184 void wpt_keep_in_circle(size_t index, const Translation2d& center,

201 void wpt_keep_in_polygon(size_t index,
202 std::span<Translation2d> field_points) {
203 for (size_t i = 0; i < field_points.size(); ++i) {
204 size_t j = (i + 1) % field_points.size();
205 wpt_constraint(
206 index,
207 PointLineRegionConstraint{
208 {0.0, 0.0}, field_points[i], field_points[j], Side::ABOVE});
209 for (const auto& bumper : bumpers) {
210 for (const auto& corner : bumper.points) {
211 wpt_constraint(
212 index, PointLineRegionConstraint{corner, field_points[i],
213 field_points[j], Side::ABOVE});
214 }
215 }

472 void wpt_constraint(size_t index, const Constraint& constraint) {
473 new_wpts(index);
474 path.waypoints.at(index).waypoint_constraints.push_back(constraint);
475 }
476
482 void sgmt_constraint(size_t from_index, size_t to_index,
483 const Constraint& constraint) {
484 assert(from_index < to_index);
485
486 new_wpts(to_index);
487 path.waypoints.at(from_index).waypoint_constraints.push_back(constraint);
488 for (size_t index = from_index + 1; index <= to_index; ++index) {
489 path.waypoints.at(index).waypoint_constraints.push_back(constraint);
490 path.waypoints.at(index).segment_constraints.push_back(constraint);
491 }
492 }
493
497 void new_wpts(size_t final_index) {
498 if (final_index < path.waypoints.size()) {
499 return;
500 }
501
502 for (size_t i = path.waypoints.size(); i <= final_index; ++i) {
503 path.waypoints.emplace_back();
504 initial_guess_points.emplace_back(std::vector<Pose2d>{{0.0, 0.0, {0.0}}});
505 if (i != 0) {
506 control_interval_counts.push_back(40);
507 }
508 }
509 }
