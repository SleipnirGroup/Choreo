181 void wpt_keep_in_circle(size_t index, const Translation2d& center,
182 double radius) {
183 for (const auto& bumper : bumpers) {
184 for (const auto& point : bumper.points) {

186 }

198 void wpt_keep_in_polygon(size_t index,
199 std::span<Translation2d> field_points) {
200 for (size_t i = 0; i < field_points.size(); ++i) {
201 size_t j = (i + 1) % field_points.size();
202 wpt_constraint(
203 index,
204 PointLineRegionConstraint{
205 {0.0, 0.0}, field_points[i], field_points[j], Side::ABOVE});
206 for (const auto& bumper : bumpers) {
207 for (const auto& corner : bumper.points) {
208 wpt_constraint(
209 index, PointLineRegionConstraint{corner, field_points[i],
210 field_points[j], Side::ABOVE});
211 }
212 }
213 }

468 void wpt_constraint(size_t index, const Constraint& constraint) {
469 new_wpts(index);
470 path.waypoints.at(index).waypoint_constraints.push_back(constraint);
471 }
472
478 void sgmt_constraint(size_t from_index, size_t to_index,
479 const Constraint& constraint) {
480 assert(from_index < to_index);
481
482 new_wpts(to_index);
483 path.waypoints.at(from_index).waypoint_constraints.push_back(constraint);
484 for (size_t index = from_index + 1; index <= to_index; ++index) {
485 path.waypoints.at(index).waypoint_constraints.push_back(constraint);
486 path.waypoints.at(index).segment_constraints.push_back(constraint);
487 }
488 }
489
493 void new_wpts(size_t final_index) {
494 if (final_index < path.waypoints.size()) {
495 return;
496 }
497
498 for (size_t i = path.waypoints.size(); i <= final_index; ++i) {
499 path.waypoints.emplace_back();
500 initial_guess_points.emplace_back(std::vector<Pose2d>{{0.0, 0.0, {0.0}}});
501 if (i != 0) {
502 control_interval_counts.push_back(40);
503 }
504 }
505 }
