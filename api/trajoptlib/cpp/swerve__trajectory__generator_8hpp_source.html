168class TRAJOPT_DLLEXPORT SwerveTrajectory {
169 public:
171 std::vector<SwerveTrajectorySample> samples;
172
173 SwerveTrajectory() = default;
174
178 explicit SwerveTrajectory(std::vector<SwerveTrajectorySample> samples)
179 : samples{std::move(samples)} {}
180
184 explicit SwerveTrajectory(const SwerveSolution& solution) {
185 double ts = 0.0;
186 for (size_t sample = 0; sample < solution.x.size(); ++sample) {
187 samples.emplace_back(
188 ts, solution.x[sample], solution.y[sample],
189 std::atan2(solution.thetasin[sample], solution.thetacos[sample]),
190 solution.vx[sample], solution.vy[sample], solution.omega[sample],
191 solution.ax[sample], solution.ay[sample], solution.alpha[sample],
192 solution.module_fx[sample], solution.module_fy[sample]);
193 ts += solution.dt[sample];
194 }
195 }
196};
