47 std::optional<PoseWithCurvature> get_point(double t) const {
48 Eigen::Vector<double, Degree + 1> polynomial_bases;
49
50 // Populate the polynomial bases
51 for (int i = 0; i <= Degree; i++) {
52 polynomial_bases(i) = std::pow(t, Degree - i);
53 }
54
55 // This simply multiplies by the coefficients. We need to divide out t some
56 // n number of times where n is the derivative we want to take.
57 Eigen::Vector<double, 6> combined = coefficients() * polynomial_bases;
58
59 double dx, dy, ddx, ddy;
60
61 // If t = 0, all other terms in the equation cancel out to zero. We can use
62 // the last x⁰ term in the equation.
63 if (t == 0.0) {
64 dx = coefficients()(2, Degree - 1);
65 dy = coefficients()(3, Degree - 1);
66 ddx = coefficients()(4, Degree - 2);
67 ddy = coefficients()(5, Degree - 2);
68 } else {
69 // Divide out t for first derivative.
70 dx = combined[2] / t;
71 dy = combined[3] / t;
72
73 // Divide out t for second derivative.
74 ddx = combined[4] / t / t;
75 ddy = combined[5] / t / t;
76 }
77
78 if (std::hypot(dx, dy) < 1e-6) {
79 return std::nullopt;
80 }
81
82 // Find the curvature.
83 const auto curvature =
84 (dx * ddy - ddx * dy) / ((dx * dx + dy * dy) * std::hypot(dx, dy));
85
86 return PoseWithCurvature{
87 {Translation2d{combined[0], combined[1]}, Rotation2d{dx, dy}},
88 curvature};
89 }
