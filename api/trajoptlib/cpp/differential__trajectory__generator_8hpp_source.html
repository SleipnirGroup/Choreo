162class TRAJOPT_DLLEXPORT DifferentialTrajectory {
163 public:
165 std::vector<DifferentialTrajectorySample> samples;
166
167 DifferentialTrajectory() = default;
168
172 explicit DifferentialTrajectory(
173 std::vector<DifferentialTrajectorySample> samples)
174 : samples{std::move(samples)} {}
175
179 explicit DifferentialTrajectory(const DifferentialSolution& solution) {
180 double ts = 0.0;
181 for (size_t sample = 0; sample < solution.x.size(); ++sample) {
182 samples.emplace_back(
183 ts, solution.x[sample], solution.y[sample], solution.heading[sample],
184 solution.vl[sample], solution.vr[sample],
185 solution.angular_velocity[sample], solution.al[sample],
186 solution.ar[sample], solution.angular_acceleration[sample],
187 solution.Fl[sample], solution.Fr[sample]);
188 ts += solution.dt[sample];
189 }
190 }
191};
